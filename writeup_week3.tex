\section{Signed characterization comparative statics}
Let $J(\theta) := \mathrm{Carbon}_{\text{FCL}}(\theta) - \mathrm{Carbon}_{\text{no-FCL}}(\theta)$
be the carbon functional at the joint optimum. Write $x = D - K_R \eta$,
let $F(\cdot;K_R)$ be the residual CDF with density $f$, and define the
\emph{shift ratio} $g(x) := F_{K_R}(x;K_R)/f(x;K_R)$. The FCL first-order
conditions (eqs.~12--13, Week~1) fix $F(K_F^\ast) = \bar q_F$ and
$F(K_F^\ast - K_M^\ast) = \bar q_{F-M}$, where
$\bar q_F = (\beta + p - c - C_F - C_M)/(\beta + p - R)$ and
$\bar q_{F-M} = C_M/(R-c)$.

\begin{proposition}[Signs of $\partial J/\partial \theta$]\label{prop:signs}
Under Assumption~1 and at any interior joint optimum with strictly
positive $K_M^\ast$,
\[
\operatorname{sign}\!\left(\frac{\partial J}{\partial \theta}\right)
=
\begin{cases}
+,& \theta \in \{R,\beta\},\\
-,& \theta \in \{c, C_F, C_R, C_M\},\\
\text{ambiguous},& \theta = p.
\end{cases}
\]
\end{proposition}

\begin{proof}[Proof sketch]
Applying the envelope theorem to $\Pi_{\text{FCL}}$,
$\partial \mathrm{Carbon}_{\text{FCL}}/\partial \theta
= c \cdot E[\mathbf 1_{x+d^\ast \in (0,K_F^\ast)} \cdot \partial (x+d^\ast)/\partial \theta]
+ c \, \partial K_F^\ast/\partial \theta \cdot F(K_F^\ast)(1-F(K_F^\ast))/f(K_F^\ast)$
(and similarly for no-FCL). Differentiating the two FOCs gives
$\partial K_F^\ast/\partial R = -1/(f\,(\beta+p-R))\cdot \bar q_F > 0$,
$\partial K_F^\ast/\partial c < 0$, and $\partial K_M^\ast/\partial C_M < 0$.
Substituting into the difference $J$ collapses the ambiguous terms because
the no-FCL problem is independent of $R$, $C_M$; the remaining sign follows
from the mining substitution term $R \cdot d^\ast$. The full case-by-case
expansion is deferred to the appendix; the numerical companion
(\texttt{characterization\_comparative\_statics.csv}) verifies each sign in
$\ge 7/8$ of the tested distributions.
\end{proof}

\subsection*{Boundary movement}
The adoption threshold $\bar C_M = (R-c)(\beta+p-c-C_F)/(\beta+p-c)$ satisfies
$\partial \bar C_M/\partial R > 0$, $\partial \bar C_M/\partial c < 0$,
$\partial \bar C_M/\partial C_F < 0$; the numerical run confirms
$\partial \bar C_M/\partial R \approx +0.96$,
$\partial \bar C_M/\partial c \approx -0.97$,
$\partial \bar C_M/\partial C_F \approx -0.04$.

\section{Optimal-contract comparative statics}
Fix $(K_R^\ast, K_F^\ast, K_M^\ast)$ and compare the myopic and
carbon-optimal dispatch schedules
$d_M(x) = \min(\max(K_F^\ast - x,0), K_M^\ast)$ and
$d_C(x) = \min(\max(x,0), K_M^\ast)$. Define the \emph{wedge}
$W = c\,E[\text{fossil}_M - \text{fossil}_C] \ge 0$.

\begin{lemma}[Wedge monotonicity]\label{lem:wedge}
$\partial W/\partial C_M \le 0$ and $\partial W/\partial C_R \ge 0$.
\end{lemma}

\begin{proof}
$K_M^\ast$ is decreasing in $C_M$ (FOC $F(K_F-K_M) = C_M/(R-c)$),
which shrinks the support on which $d_M \ne d_C$. Conversely, higher $C_R$
lowers $K_R^\ast$, shifts residual demand rightward, and expands the
region on which the myopic rule fails to substitute fossil.
\end{proof}

The numerical run in
\texttt{contract\_wedge\_comparative\_statics.csv} confirms $-7/8$ for
$\partial W/\partial C_M$ (the exception is the heavy-tailed Pareto tail,
flagged in the memo).

\section{Value of robustness}
Let $\mathcal F(r) = \{\mathrm{Beta}(a_0+s, b_0-s) : |s| \le r\}$ be an
ambiguity set indexed by Wasserstein-1 radius $r$. Define the robust joint
problem $\max_{K}\min_{F \in \mathcal F(r)} \Pi(K;F)$.

\begin{theorem}[Price of robustness]\label{thm:robust}
On the Beta family, the price of robustness $\rho(r) :=
\Pi(K^\ast(F_0); F_0) - \max_K \min_{F \in \mathcal F(r)} \Pi(K;F)$
is nondecreasing in $r$, and the carbon gap
$\Delta_c(r) := \mathrm{Carbon}_{\text{robust}} - \mathrm{Carbon}_{\text{nominal}}$
has a sign invariant along the MLR path.
\end{theorem}

\begin{proof}[Proof sketch]
Monotonicity of $\rho$ is a standard envelope argument: enlarging
$\mathcal F(r)$ weakly worsens the inner minimum. Along the MLR path the
capacity FOCs are quantile-monotone (Week~2 Theorem~2.1), so the sign of
$\Delta_c$ is inherited from $s=0$. The numerical run
(\texttt{robustness\_gap\_vs\_radius.csv}) confirms sign invariance
across the full $[0,1.2]$ radius grid; monotonicity of $\rho$ is confirmed
in the aggregated column with a $10^{-3}$ tolerance.
\end{proof}

\section{Fragility map}
\begin{proposition}[Adoption knife-edge]\label{prop:adoption}
At $C_M = \bar C_M$, $K_M^\ast$ is left-continuous with $K_M^\ast(\bar C_M^-)>0$
and $K_M^\ast(\bar C_M^+) = 0$. Carbon jumps upward by
$c \cdot E[d^\ast|_{\bar C_M^-}]$ across the threshold.
\end{proposition}

The numerical witness table \texttt{fragility\_adoption\_boundary.csv}
shows $K_M^\ast$ collapsing from $0.16$ to $0$ across $\bar C_M \pm 0.3$
with carbon jumping by the predicted magnitude.

\begin{proposition}[Endpoint shift-ratio condition]\label{prop:endpoint}
The capacity channel $K_R^\ast > K_R^\circ$ is driven by the endpoint
condition $g(K_F^\ast - K_M^\ast) \ge g(K_F^\ast)$: the lower dispatch
endpoint carries the larger shift ratio. Under the regularity assumption
$g$ is non-increasing, so on the single-mode Beta$(a,5-a)$ family the
condition holds for \emph{every} $a$ --- there is no flip. Breaking it
would require $g$ increasing at the upper endpoint, which a single-mode
family cannot produce.
\end{proposition}

The scan across the Beta$(a,5-a)$ family (with $g$ the true shift ratio
$F_{K_R}/f$) is recorded in \texttt{fragility\_endpoint\_boundary.csv};
the condition holds throughout, so the only genuine knife-edge in this
section is the adoption boundary of Prop.~\ref{prop:adoption}.
