\section{Model recap}
Residual demand is
\[
x \;=\; D - K_R\,\eta, \qquad D\in[0,\bar D],\; \eta\in[0,1],
\]
with CDF $F(\,\cdot\,;K_R)$. The utility chooses capacities $(K_F,K_R,K_M)$ and
dispatches the flexible computing load by the threshold rule
$d(x)=\min\{(K_F-x)_+,K_M\}$ (the optimal within-period contract; Role~1).
We treat the per-MWh computing margin $R$ as exogenous, consistent with Role~1's
baseline; the miner-side market structure (oligopoly demand, mining price) is a
B-workstream / miner-side object and is \emph{outside A1's scope}, so it does not
enter the comparative statics below.

\paragraph{Assumption 1 (interior regime; identical to Role~1/Role~4).}
\[
R>c+C_M,\qquad R<c+C_F+C_M,\qquad \beta+p>c+C_F,\qquad \beta+p>R.
\]

\section{No-FCL benchmark}
Without FCL the planner solves
\[
\max_{K_F,K_R}\ p\,\mathbb E[\bar D-(x-K_F)_+]
-c\,\mathbb E[\min(x_+,K_F)]
-(\beta+p)\mathbb E[(x-K_F)_+]
-C_F K_F-C_R K_R.
\]
First-order conditions give
\begin{align}
F(K_F^{\circ};K_R^{\circ}) &= 1-\frac{C_F}{\beta+p-c}, \label{eq:KFno}\\
\mathbb E[\eta\mid x>0]\bigl(c\,F(0)+(\beta+p)(1-F(K_F^{\circ}))\bigr) &=C_R. \label{eq:KRno}
\end{align}

\section{Joint optimum with FCL}
With FCL the FCL dispatch is $d(x)=\min\{(K_F-x)_+,K_M\}$. The planner's FOCs (eqs.~12,13,15 of the paper) are
\begin{align}
F(K_F^{\star};K_R^{\star}) &= \frac{\beta+p-c-C_F-C_M}{\beta+p-R}, \label{eq:KF}\\
F(K_F^{\star}-K_M^{\star};K_R^{\star}) &= \frac{C_M}{R-c}, \label{eq:KM}
\end{align}
together with a renewable FOC of the same form as \eqref{eq:KRno} but with the FCL dispatch added in the inner region.

\section{Proposition 1 (distribution-free)}
\begin{proposition}\label{prop:1}
Under Assumption 1 and $C_M<\bar C_M$ with
\[
\bar C_M \;=\; (R-c)\,\frac{\beta+p-c-C_F}{\beta+p-c},
\]
introducing FCL strictly increases the utility's profit and strictly decreases blackout probability, for any continuous residual-demand distribution $F$.
\end{proposition}

\begin{proof}
\emph{Profit.} The FCL planner's program nests the no-FCL one ($K_M=0$ is feasible). Optimality therefore weakly raises profit; the strict inequality follows because at $K_M=0$ the gradient of the Lagrangian in $K_M$ equals $(R-c)\bigl[1-F(K_F^{\circ})\bigr]-C_M$, which is positive iff $C_M<\bar C_M$, so the optimum has $K_M^{\star}>0$ and profit strictly rises.

\emph{Blackout.} Comparing \eqref{eq:KF} and \eqref{eq:KFno}: the right-hand side of \eqref{eq:KF} exceeds that of \eqref{eq:KFno} in the interior regime, so the target quantile is higher and $K_F^{\star}>K_F^{\circ}$ when $K_R$ is held fixed. Lemma~\ref{lem:sub} only reinforces this through $K_R^{\star}\ge K_R^{\circ}$. Hence $\Pr(x>K_F^{\star})<\Pr(x>K_F^{\circ})$.
\end{proof}

\begin{lemma}[Capacity substitution]\label{lem:sub}
In the interior regime, $K_R^{\star}\ge K_R^{\circ}$, with strict inequality whenever $K_M^{\star}>0$.
\end{lemma}

\begin{proof}
Comparing renewable FOCs: the FCL version adds a strictly positive term $(R-c)\mathbb E[\eta\,\mathbf 1\{0<x<K_F-K_M\}]$ to the marginal benefit of renewable capacity, while the blackout term is (weakly) smaller because $K_F^{\star}\ge K_F^{\circ}$. Net marginal benefit of $K_R$ rises, so the optimal $K_R$ rises.
\end{proof}

\section{Carbon decomposition (eq.~17) --- two forces, matching Role~1}
We adopt Role~1's decomposition verbatim to avoid any ambiguity. The
\emph{reduction} in carbon from introducing FCL (no-FCL emissions minus with-FCL
emissions) splits into two opposing forces:
\begin{equation}
\underbrace{c\,\mathbb E[\text{fossil}^{\circ}]-c\,\mathbb E[\text{fossil}^{\star}]}_{\Delta\text{Carbon (reduction)}}
\;=\;
\underbrace{\text{capacity-substitution effect}}_{\ge 0\ \text{when}\ K_R^{\star}>K_R^{\circ}\ (\le 0\ \text{otherwise})}
\;-\;
\underbrace{\text{utilization effect}}_{>0,\ \text{distribution-free}} . \label{eq:carbon}
\end{equation}
The \textbf{utilization effect} is Role~1's Term~(II): running computing on spare
fossil strictly \emph{raises} fossil output in every state, so it is always
positive and reduces the saving. The \textbf{capacity-substitution effect} is
Role~1's Term~(I): the larger renewable fleet $K_R^{\star}>K_R^{\circ}$
(Lemma~\ref{lem:sub}) shifts residual demand left and \emph{lowers} fossil output.
Carbon falls iff the second beats the first --- exactly Role~1's condition~(C).
Our numerics below confirm both signs and map where the net is negative.

\section{Section 5.3 (Case C, short run)}
\begin{corollary}\label{cor:caseC}
If $(K_F^{(0)},K_R^{(0)})$ are exogenously fixed and only $K_M$ is optimised, then
the carbon reduction $\Delta\text{Carbon}\le 0$ --- FCL weakly \emph{increases}
emissions (Role~1, \S5.3 / eq.~30). Equality holds iff
$F(K_F^{(0)};K_R^{(0)})\le C_M/(R-c)$, i.e.\ $K_M^{\star}=0$.
\end{corollary}

\begin{proof}
Capacities are frozen, so the capacity-substitution effect in \eqref{eq:carbon}
vanishes identically. Only the positive utilization effect remains, so the
reduction is $\le 0$.
\end{proof}

\section{Sync notes for Role 1}
\begin{itemize}
\item Profit and blackout halves of Proposition~\ref{prop:1} verified distribution-free across 8 families whenever $C_M<\bar C_M$.
\item Utilization effect (Role~1 Term~(II)) is positive --- it \emph{raises} fossil output --- in every scenario, matching the distribution-free sign.
\item Capacity-substitution channel $K_R^{\star}>K_R^{\circ}$ holds in 32/32 scenarios --- this is the binding driver of the carbon sign.
\item Left-skewed $F$ with high $C_R$ reverses the carbon sign; this is the boundary region for Role~1's general statement.
\end{itemize}
